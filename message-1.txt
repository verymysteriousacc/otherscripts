local Connections = {}

local function Connect(signal, callback)
	local c = signal:Connect(callback)
	table.insert(Connections, c)
	return c
end

local function DisconnectAll()
	for _, c in ipairs(Connections) do
		pcall(function()
			c:Disconnect()
		end)
	end

	table.clear(Connections)
end

local AccentObjects = {}
local CornerObjects = {}
local SurfaceObjects = {}

local function RegisterAccent(object, property)
	table.insert(AccentObjects, {
		Object = object,
		Property = property,
	})
end

local function RegisterCorner(corner)
	table.insert(CornerObjects, corner)
end

local function RegisterSurface(object)
	table.insert(SurfaceObjects, object)
end

local function AddCorner(object, radius)
	local corner = Instance.new("UICorner")

	corner.CornerRadius = UDim.new(
		0,
		radius or Config.CornerRadius
	)

	corner.Parent = object

	RegisterCorner(corner)

	return corner
end

local function RefreshTheme()
	local purple = Purple()

	for _, data in ipairs(AccentObjects) do
		if data.Object and data.Object.Parent then
			pcall(function()
				data.Object[data.Property] = purple
			end)
		end
	end

	for _, corner in ipairs(CornerObjects) do
		if corner and corner.Parent then
			corner.CornerRadius = UDim.new(
				0,
				math.floor(Config.CornerRadius)
			)
		end
	end

	for _, object in ipairs(SurfaceObjects) do
		if object and object.Parent then
			object.BackgroundTransparency =
				Config.PanelTransparency
		end
	end
end

local function TweenObject(
	object,
	duration,
	properties,
	style,
	direction
)
	if not object then
		return
	end

	if not Config.Animations then
		for property, value in pairs(properties) do
			object[property] = value
		end
		return
	end

	local speed = math.max(
		Config.AnimationSpeed,
		0.05
	)

	TweenService:Create(
		object,
		TweenInfo.new(
			duration / speed,
			style or Enum.EasingStyle.Quint,
			direction or Enum.EasingDirection.Out
		),
		properties
	):Play()
end

local function UpdateScale()
	if not Camera then
		return
	end

	local viewport = Camera.ViewportSize

	local widthScale =
		math.max(viewport.X - 14, 360) / 620

	local heightScale =
		math.max(viewport.Y - 14, 260) / 430

	local auto =
		math.min(
			widthScale,
			heightScale,
			1
		)

	auto = math.max(auto, 0.5)

	MainScale.Scale =
		auto * Config.UIScale
end

Connect(
	Camera:GetPropertyChangedSignal("ViewportSize"),
	UpdateScale
)

UpdateScale()

local Tabs = {}

local function SelectPage(target)

	for _, data in pairs(Tabs) do

		local active =
			data.Page == target

		data.Page.Visible =
			active

		TweenObject(
			data.Button,
			0.15,
			{
				BackgroundTransparency =
					active
					and 0
					or 1
			}
		)

		TweenObject(
			data.Label,
			0.15,
			{
				TextColor3 =
					active
					and COLOR.PurpleLight
					or COLOR.PurpleDark
			}
		)

		TweenObject(
			data.Line,
			0.15,
			{
				BackgroundTransparency =
					active
					and 0
					or 1
			}
		)

	end
end

local function GetTargetPart(character)

	if not character then
		return nil
	end

	local selected =
		character:FindFirstChild(
			Config.TargetPart
		)

	if selected
		and selected:IsA("BasePart") then

		return selected
	end

	for _, name in ipairs({
		"Head",
		"HumanoidRootPart",
		"UpperTorso",
		"LowerTorso",
	}) do

		local fallback =
			character:FindFirstChild(
				name
			)

		if fallback
			and fallback:IsA("BasePart") then

			return fallback
		end

	end

	return nil
end

local function Alive(player)

	local character =
		player.Character

	local humanoid =
		character
		and character:
			FindFirstChildOfClass(
				"Humanoid"
			)

	return character
		and humanoid
		and humanoid.Health > 0
end

local function IsTeammate(player)

	if not Config.TeamCheck then
		return false
	end

	return LocalPlayer.Team
		and player.Team
		and LocalPlayer.Team ==
			player.Team
end

--==============================================================
-- WALL CHECK
--==============================================================

local function CanSee(player, part)

	if not Config.WallCheck then
		return true
	end

	local localCharacter =
		GetCharacter()

	local params =
		RaycastParams.new()

	params.FilterType =
		Enum.RaycastFilterType.Exclude

	params.IgnoreWater =
		true

	if localCharacter then

		params.FilterDescendantsInstances =
			{
				localCharacter
			}

	end

	local origin =
		Camera.CFrame.Position

	local direction =
		part.Position - origin

	local result =
		workspace:Raycast(
			origin,
			direction,
			params
		)

	if not result then
		return true
	end

	return result.Instance:
		IsDescendantOf(
			player.Character
		)
end

--==============================================================
-- TARGET FINDER
--==============================================================

local function FindTarget()

	local center =
		Vector2.new(
			Camera.ViewportSize.X * 0.5,
			Camera.ViewportSize.Y * 0.5
		)

	local bestPlayer = nil
	local bestPart = nil

	local bestMetric =
		Config.TargetPriority ==
		"Farthest"
		and -math.huge
		or math.huge

	for _, player in ipairs(
		Players:GetPlayers()
	) do

		if player == LocalPlayer then
			continue
		end

		if Config.AliveCheck
			and not Alive(player) then

			continue
		end

		if IsTeammate(player) then
			continue
		end

		local character =
			player.Character

		local part =
			GetTargetPart(
				character
			)

		if not character
			or not part then

			continue
		end

		local distance =
			(
				part.Position
				- Camera.CFrame.Position
			).Magnitude

		if distance >
			Config.AimDistance then

			continue
		end

		local screen,
			onScreen =
			Camera:WorldToViewportPoint(
				part.Position
			)

		if not onScreen
			or screen.Z <= 0 then

			continue
		end

		local screenDistance =
			(
				Vector2.new(
					screen.X,
					screen.Y
				)
				- center
			).Magnitude

		if screenDistance >
			Config.FOV then

			continue
		end

		if not CanSee(
			player,
			part
		) then

			continue
		end

		if Config.TargetPriority ==
			"Nearest" then

			if screenDistance <
				bestMetric then

				bestMetric =
					screenDistance

				bestPlayer =
					player

				bestPart =
					part
			end

		else

			if distance >
				bestMetric then

				bestMetric =
					distance

				bestPlayer =
					player

				bestPart =
					part
			end
		end
	end

	return bestPlayer, bestPart
end

local function TargetValid(player)

	if not player then
		return false
	end

	local character =
		player.Character

	local humanoid =
		character
		and character:
			FindFirstChildOfClass(
				"Humanoid"
			)

	local part =
		GetTargetPart(
			character
		)

	if not character
		or not humanoid
		or not part then

		return false
	end

	if Config.AliveCheck
		and humanoid.Health <= 0 then

		return false
	end

	if IsTeammate(player) then
		return false
	end

	local distance =
		(
			part.Position
			- Camera.CFrame.Position
		).Magnitude

	if distance >
		Config.AimDistance then

		return false
	end

	local screen,
		onScreen =
		Camera:WorldToViewportPoint(
			part.Position
		)

	if not onScreen
		or screen.Z <= 0 then

		return false
	end

	local center =
		Vector2.new(
			Camera.ViewportSize.X * 0.5,
			Camera.ViewportSize.Y * 0.5
		)

	local screenDistance =
		(
			Vector2.new(
				screen.X,
				screen.Y
			)
			- center
		).Magnitude

	if screenDistance >
		Config.FOV then

		return false
	end

	return CanSee(
		player,
		part
	)
end

local HoldAim = false
local ToggleAim = false

Connect(
	UserInputService.InputBegan,
	function(input, processed)

		if processed then
			return
		end

		if input.KeyCode ==
			Enum.KeyCode.RightShift then

			if not Config.Destroyed then

				if Config.Minimized then
					Config.Minimized = false
					Body.Visible = true
				end

				Config.Visible =
					not Config.Visible

				Main.Visible =
					Config.Visible
			end
		end

		if input.KeyCode ==
			Config.AimKey then

			if Config.AimMode ==
				"Hold" then

				HoldAim = true

			elseif Config.AimMode ==
				"Toggle" then

				ToggleAim =
					not ToggleAim
			end
		end
	end
)

Connect(
	UserInputService.InputEnded,
	function(input)

		if input.KeyCode ==
			Config.AimKey
			and Config.AimMode ==
			"Hold" then

			HoldAim = false
		end
	end
)

local function AimActive()

	if not Config.Aimbot then
		return false
	end

	if Config.AimMode ==
		"Always" then

		return true
	end

	if Config.AimMode ==
		"Hold" then

		return HoldAim
	end

	if Config.AimMode ==
		"Toggle" then

		return ToggleAim
	end

	return false
end

local function FlyStep()

	if not Config.Fly then
		return
	end

	if not FlyVelocity
		or not FlyGyro then

		StartFly()
	end

	if not FlyVelocity
		or not FlyGyro then

		return
	end

	local movement =
		Vector3.zero

	if UserInputService:IsKeyDown(
		Enum.KeyCode.W
	) then

		movement +=
			Camera.CFrame.LookVector
	end

	if UserInputService:IsKeyDown(
		Enum.KeyCode.S
	) then

		movement -=
			Camera.CFrame.LookVector
	end

	if UserInputService:IsKeyDown(
		Enum.KeyCode.D
	) then

		movement +=
			Camera.CFrame.RightVector
	end

	if UserInputService:IsKeyDown(
		Enum.KeyCode.A
	) then

		movement -=
			Camera.CFrame.RightVector
	end

	if UserInputService:IsKeyDown(
		Enum.KeyCode.Space
	) then

		movement +=
			Vector3.yAxis
	end

	if UserInputService:IsKeyDown(
		Enum.KeyCode.LeftControl
	) then

		movement -=
			Vector3.yAxis
	end

	if movement.Magnitude > 0 then
		movement =
			movement.Unit
	end

	FlyVelocity.Velocity =
		movement
		* Config.FlySpeed

	FlyGyro.CFrame =
		Camera.CFrame
end

local ESPObjects = {}

local function RemoveESP(player)

	local data =
		ESPObjects[player]

	if not data then
		return
	end

	for _, object in pairs(data) do

		if object
			and object.Parent then

			object:Destroy()
		end
	end

	ESPObjects[player] =
		nil
end

local function CreateESP(player)

	if player == LocalPlayer
		or ESPObjects[player] then

		return
	end

	local character =
		player.Character

	local head =
		character
		and character:
			FindFirstChild(
				"Head"
			)

	if not character
		or not head then

		return
	end

	local highlight =
		Instance.new(
			"Highlight"
		)

	highlight.Name =
		"AnarchyESP"

	highlight.Adornee =
		character

	highlight.DepthMode =
		Enum.HighlightDepthMode
			.AlwaysOnTop

	highlight.FillTransparency =
		Config.ESPFill
		and 0.84
		or 1

	highlight.OutlineTransparency =
		Config.ESPOutline
		and 0.08
		or 1

	highlight.FillColor =
		Purple()

	highlight.OutlineColor =
		Purple()

	highlight.Parent =
		character

	local billboard =
		Instance.new(
			"BillboardGui"
		)

	billboard.Name =
		"AnarchyESPInfo"

	billboard.Adornee =
		head

	billboard.Size =
		UDim2.fromOffset(
			220,
			48
		)

	billboard.StudsOffset =
		Vector3.new(
			0,
			2.7,
			0
		)

	billboard.AlwaysOnTop =
		true

	billboard.MaxDistance =
		Config.ESPDistance

	billboard.Parent =
		head

	local label =
		Instance.new(
			"TextLabel"
		)

	label.Size =
		UDim2.fromScale(
			1,
			1
		)

	label.BackgroundTransparency =
		1

	label.Font =
		Enum.Font.GothamBold

	label.TextSize =
		9.5

	label.TextWrapped =
		true

	label.TextYAlignment =
		Enum.TextYAlignment.Center

	label.TextColor3 =
		COLOR.PurpleLight

	label.TextStrokeTransparency =
		0.75

	label.Parent =
		billboard

	ESPObjects[player] =
		{
			Highlight = highlight,
			Billboard = billboard,
			Label = label,
		}
end

local function UpdateESP()

	if not Config.ESP then

		for player in pairs(
			ESPObjects
		) do

			RemoveESP(player)
		end

		return
	end

	for _, player in ipairs(
		Players:GetPlayers()
	) do

		if player == LocalPlayer then
			continue
		end

		local character =
			player.Character

		local humanoid =
			character
			and character:
				FindFirstChildOfClass(
					"Humanoid"
				)

		local root =
			character
			and character:
				FindFirstChild(
					"HumanoidRootPart"
				)

		if not character
			or not humanoid
			or not root then

			RemoveESP(player)
			continue
		end

		if humanoid.Health <= 0 then
			RemoveESP(player)
			continue
		end

		if Config.ESPTeamCheck
			and IsTeammate(player) then

			RemoveESP(player)
			continue
		end

		local distance =
			(
				root.Position
				- Camera.CFrame.Position
			).Magnitude

		if distance >
			Config.ESPDistance then

			RemoveESP(player)
			continue
		end

		if not ESPObjects[player] then
			CreateESP(player)
		end

		local data =
			ESPObjects[player]

		if data then

			local parts = {}

			if Config.ESPNames then
				table.insert(
					parts,
					player.DisplayName
				)
			end

			if Config.ESPDistanceText then
				table.insert(
					parts,
					math.floor(distance)
					.. " st"
				)
			end

			if Config.ESPHealth then
				table.insert(
					parts,
					"HP "
					.. math.floor(
						humanoid.Health
					)
				)
			end

			data.Label.Text =
				table.concat(
					parts,
					"\n"
				)

			data.Highlight.FillTransparency =
				Config.ESPFill
				and 0.84
				or 1

			data.Highlight.OutlineTransparency =
				Config.ESPOutline
				and 0.08
				or 1

			data.Highlight.FillColor =
				Purple()

			data.Highlight.OutlineColor =
				Purple()

			data.Billboard.MaxDistance =
				Config.ESPDistance
		end
	end
end

--==============================================================
-- DRAGGING
--==============================================================

local Dragging = false
local DragInput = nil
local DragStart = nil
local StartPosition = nil

DragArea.InputBegan:Connect(
	function(input)

		if input.UserInputType
			== Enum.UserInputType.MouseButton1
			or input.UserInputType
			== Enum.UserInputType.Touch then

			Dragging = true
			DragStart = input.Position
			StartPosition = Main.Position
		end
	end
)

DragArea.InputChanged:Connect(
	function(input)

		if input.UserInputType
			== Enum.UserInputType.MouseMovement
			or input.UserInputType
			== Enum.UserInputType.Touch then

			DragInput = input
		end
	end
)

Connect(
	UserInputService.InputChanged,
	function(input)

		if not Dragging then
			return
		end

		if input ~= DragInput then
			return
		end

		local delta =
			input.Position
			- DragStart

		Main.Position =
			UDim2.new(
				StartPosition.X.Scale,
				StartPosition.X.Offset
					+ delta.X,

				StartPosition.Y.Scale,
				StartPosition.Y.Offset
					+ delta.Y
			)
	end
)

Connect(
	UserInputService.InputEnded,
	function(input)

		if input.UserInputType
			== Enum.UserInputType.MouseButton1
			or input.UserInputType
			== Enum.UserInputType.Touch then

			Dragging = false
			DragInput = nil
		end
	end
)

--==============================================================
-- MAIN LOOP
--==============================================================

local CurrentTarget = nil

local FPS = 60
local FPSTimer = 0
local FrameCounter = 0

local ESPTimer = 0

Connect(
	RunService.RenderStepped,
	function(deltaTime)

		if Config.Destroyed then
			return
		end

		-- FPS CALCULATION
		-- Kept for the Server page.
		FPSTimer += deltaTime
		FrameCounter += 1

		if FPSTimer >= 0.5 then

			FPS =
				FrameCounter
				/ FPSTimer

			FrameCounter = 0
			FPSTimer = 0
		end

		-- FOV
		local viewport =
			Camera.ViewportSize

		FOVCircle.Position =
			UDim2.fromOffset(
				viewport.X * 0.5,
				viewport.Y * 0.5
			)

		FOVCircle.Size =
			UDim2.fromOffset(
				Config.FOV * 2,
				Config.FOV * 2
			)

		FOVCorner.CornerRadius =
			UDim.new(
				1,
				0
			)

		FOVStroke.Thickness =
			Config.FOVThickness

		FOVStroke.Transparency =
			Config.FOVTransparency

		FOVCircle.Visible =
			Config.Aimbot
			and Config.ShowFOV

		-- AIM
		if AimActive() then

			local valid =
				Config.StickyTarget
				and CurrentTarget
				and TargetValid(
					CurrentTarget
				)

			if not valid then

				local player =
					FindTarget()

				CurrentTarget =
					player
			end

			if CurrentTarget then

				local part =
					GetTargetPart(
						CurrentTarget.Character
					)

				if part then

					ShowTarget(
						CurrentTarget
					)

					local targetPosition =
						part.Position

					if Config.Prediction >
						0 then

						targetPosition +=
							part.AssemblyLinearVelocity
							* Config.Prediction
					end

					local cameraPosition =
						Camera.CFrame.Position

					local desired =
						CFrame.lookAt(
							cameraPosition,
							targetPosition
						)

					local alpha =
						1 - math.exp(
							-(
								Config.Smoothness
								* 0.55
							)
							* deltaTime
						)

					alpha =
						math.clamp(
							alpha,
							0,
							1
						)

					Camera.CFrame =
						Camera.CFrame:Lerp(
							desired,
							alpha
						)

				else

					CurrentTarget =
						nil

					ShowTarget(nil)
				end

			else

				ShowTarget(nil)
			end

		else

			CurrentTarget =
				nil

			ShowTarget(nil)
		end

		-- MOVEMENT
		ApplyMovement()

		NoclipStep()

		FlyStep()

		-- VISUALS
		ApplyFullBright()

		ApplyCamera()

		-- ESP
		ESPTimer += deltaTime

		if ESPTimer >= 0.08 then

			ESPTimer = 0

			UpdateESP()
		end

		-- SERVER
		UpdateServerInfo()
	end
)
